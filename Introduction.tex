The equation
\begin{equation}\label{eq:main}
x^3+y^3=x^2+kxy+y^2,\qquad k\in\mathbb Z,
\end{equation}
with $k=42$ is Problem~1 of the Balkan Mathematical Olympiad (BMO) 2017~\cite{bmo}, where only \emph{positive} integer solutions were required. In this paper we study all integer solutions of~\eqref{eq:main} for an arbitrary integer parameter $k$, and we write
\[
N_k=\#\{(x,y)\in\mathbb Z^2:\ x^3+y^3=x^2+kxy+y^2\}.
\]
For every $k$ the pairs $(0,0)$, $(1,0)$ and $(0,1)$ satisfy~\eqref{eq:main}; we call them the \emph{trivial solutions}, so $N_k\ge 3$ for all $k$. For $k=42$ one finds $N_{42}=8$: besides the trivial solutions there are $(1,7)$, $(7,1)$, $(22,22)$ (the answer to the olympiad problem) and the two solutions $(-6,1)$, $(1,-6)$ with a negative coordinate.

\medskip
\noindent\textbf{Summary of results.} For $k=-1$ and $k=2$ the equation has infinitely many integer solutions (see below). For every other $k$ the number $N_k$ is finite and is bounded by a divisor-function expression (Theorem~2), which yields a linear bound $N_k\le c|k|$ (Theorem~4). On the other hand, $N_k$ is unbounded as $k$ varies (Theorem~1), while $N_k=4$ for infinitely many $k$ (Theorem~3).

\subsection{Small values of \texorpdfstring{$k$}{k}}
We first treat the values $k=0,-1,1,2$ by elementary means.
\begin{enumerate}
    \item \textbf{$k=0$.} The equation is $x^3+y^3=x^2+y^2$. Put $Q=x^2-xy+y^2$, so that $x^3+y^3=(x+y)Q$. If $xy=0$, we get the solutions $(0,0)$, $(1,0)$, $(0,1)$. Let $xy\neq 0$. Since $Q\mid x^3+y^3$ we have $Q\mid x^2+y^2$, hence $Q\mid (x^2+y^2)-Q=xy$, and so $Q\le |xy|$. But if $xy<0$ then $Q=x^2+y^2+|xy|>|xy|$, and if $xy>0$ then $Q-xy=(x-y)^2\ge 0$. Therefore $xy>0$ and $x=y$, and the equation becomes $2x^3=2x^2$, i.e.\ $x=y=1$.
    \\
    \textbf{Answer:}
\[
(x,y) = (0,0),\ (1,0),\ (0,1),\ (1,1).
\]
    \item \textbf{$k=-1$.} The equation is $x^3+y^3=x^2-xy+y^2$, that is, $(x+y)Q=Q$ with $Q$ as above, or $(x+y-1)Q=0$. Hence either $x+y=1$ or $Q=0$, which over the integers means $x=y=0$.
    \\
    \textbf{Answer:}
\[
(x,y) = (0,0),\quad (x,1-x)\ \ (x\in\mathbb Z).
\]
    \item \textbf{$k=1$.} The equation is $x^3+y^3=x^2+xy+y^2$. Let
\[
s = x + y, \quad p = xy.
\]
Then
\[
x^3 + y^3 = (x+y)^3 - 3xy(x+y) = s^3 - 3ps,
\]
\[
x^2 + xy + y^2 = (x+y)^2 - xy = s^2 - p.
\]
Substituting, we get $s^3 - 3ps = s^2 - p$, that is,
\[
p(3s - 1) = s^2(s - 1).
\]
Since $\gcd(3s-1,s)=1$ (a common divisor divides $(3s-1)-3s=-1$), also $\gcd(3s-1,s^2)=1$, and therefore $3s-1\mid s-1$. Consequently
\[
3s-1 \mid (3s-1)-3(s-1)=2,
\]
so $3s-1\in\{-2,-1,1,2\}$. Only $3s-1=-1$ and $3s-1=2$ give integer values of $s$, namely $s=0$ and $s=1$.

\textbf{Case 1:} $s = 0$. Then $-p = 0$, so $p=0$, and $x + y = 0$, $xy = 0$ give $x = y = 0$.

\textbf{Case 2:} $s = 1$. Then $2p = 0$, so $p=0$, and $x + y = 1$, $xy = 0$ give $(x,y) = (1,0)$ or $(0,1)$.

\textbf{Answer:}
\[
(x,y) = (0,0),\ (1,0),\ (0,1).
\]
    \item \textbf{$k=2$.} The equation becomes
\[
x^3+y^3=x^2+2xy+y^2=(x+y)^2.
\]
Every pair with $x+y=0$ is a solution, so there are infinitely many integer solutions. If $x+y\ne 0$, dividing by $x+y$ gives $x^2-xy+y^2=x+y$, that is, $x^2+y^2=xy+x+y$. Doubling both sides, we obtain
\[
(x-y)^2+(x-1)^2+(y-1)^2=2.
\]
A sum of three squares equals $2$ only if two of the squares are $1$ and the third is $0$. If $x=y$, then $(x-1)^2=1$, so $x=y=2$ (the case $x=y=0$ has $x+y=0$). If $x=1$, then $(y-1)^2=1$, so $y=0$ or $y=2$. If $y=1$, similarly $x=0$ or $x=2$.
    \\
    \textbf{Answer:}
\[
(x,y) = (x,-x)\ \ (x\in\mathbb Z),\quad (1,0),\ (0,1),\ (1,2),\ (2,1),\ (2,2).
\]
\end{enumerate}
