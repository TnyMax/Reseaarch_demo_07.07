\begin{thmbox}
\textbf{Lemma 2.2.} For every prime $p$ the function $a\mapsto f(p,a)=\dfrac{a+1}{p^{a/3}}$ is bounded on the non-negative integers. In particular, $C_p:=\sup_{a\ge 0}f(p,a)$ is finite, and $C_p\ge f(p,0)=1$.
\end{thmbox}

\begin{proof}
For every $a\ge 0$,
\[
\frac{f(p,a+1)}{f(p,a)}=\frac{a+2}{(a+1)\,p^{1/3}}=\frac{1+\frac{1}{a+1}}{p^{1/3}} .
\]
Since $p^{1/3}>1$, there is an $a_0$ such that $1+\frac{1}{a_0+1}<p^{1/3}$, and then the ratio is less than $1$ for all $a\ge a_0$. Hence $f(p,\cdot)$ is decreasing from $a_0$ onwards, and
\[
\sup_{a\ge 0}f(p,a)=\max_{0\le a\le a_0}f(p,a)<\infty. \qedhere
\]
\end{proof}

\noindent\textit{Remark.} For the primes $p<27$ the maxima can be found directly:
$C_2=f(2,3)=2$, $C_3=f(3,2)\approx 1.442$, $C_5=f(5,1)\approx 1.170$, $C_7=f(7,1)\approx 1.046$, and $C_p=1$ for $p=11,13,17,19,23$. Figure~\ref{fig:f2} illustrates the case $p=2$.
